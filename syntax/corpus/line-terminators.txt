================================================================================
a CR LF ends the line before a parenthesis
================================================================================

let f = {
    let a = g
    (x)
}

--------------------------------------------------------------------------------

(source-file
  (let-definition
    (identifier)
    (block
      (let-binding (identifier) (variable (identifier)))
      (parenthesized-expression (variable (identifier))))))

================================================================================
a CR on its own is whitespace, so the parenthesis after it applies
================================================================================

let f = {    let a = g    (x)    a}

--------------------------------------------------------------------------------

(source-file
  (let-definition
    (identifier)
    (block
      (let-binding
        (identifier)
        (application
          (variable (identifier))
          (argument-list (variable (identifier)))))
      (variable (identifier)))))

================================================================================
a CR on its own is a character of a string literal
================================================================================

let a = "xy"

--------------------------------------------------------------------------------

(source-file
  (let-definition (identifier) (string-literal)))

================================================================================
a CR LF ends a string literal
:error
================================================================================

let a = "x
y"

--------------------------------------------------------------------------------

================================================================================
a CR on its own does not end a comment
================================================================================

let a = 1 // clet b = 2

--------------------------------------------------------------------------------

(source-file
  (let-definition (identifier) (integer-literal)))

================================================================================
a CR LF ends a comment
================================================================================

let a = 1 // c
let b = 2

--------------------------------------------------------------------------------

(source-file
  (let-definition (identifier) (integer-literal))
  (let-definition (identifier) (integer-literal)))
